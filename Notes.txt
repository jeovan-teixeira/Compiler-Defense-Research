Compiler Defense Configs

No Flags
gcc vulnerable_server.c -o server_noFlags

Baseline (No Hardening)
gcc -O0 -fno-stack-protector -no-pie -Wl,-z,execstack -Wl,-z,norelro vulnerable_server.c -o server_baseline

Stack Canary Only
gcc -O0 -fstack-protector-strong -no-pie -Wl,-z,execstack -Wl,-z,norelro vulnerable_server.c -o server_canary

NX Only
gcc -O0 -fno-stack-protector -no-pie -Wl,-z,noexecstack -Wl,-z,norelro vulnerable_server.c -o server_nx

PIE + NX
gcc -O0 -fno-stack-protector -fPIE -pie -Wl,-z,noexecstack -Wl,-z,norelro vulnerable_server.c -o server_pie_nx

Canary + PIE + NX + RELRO
gcc -O0 -fstack-protector-strong -fPIE -pie -Wl,-z,noexecstack -Wl,-z,relro,-z,now vulnerable_server.c -o server_hardened

Canary + PIE + NX + RELRO + _FORTIFY_SOURCE
gcc -O2 -fstack-protector-strong -fPIE -pie -Wl,-z,noexecstack -Wl,-z,relro,-z,now -D_FORTIFY_SOURCE=2 vulnerable_server.c -o server_full
